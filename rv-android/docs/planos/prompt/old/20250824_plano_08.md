## **Performance Monitoring Integration**

### **Use Existing PerformanceMonitor**
```python
from rv_android_core.util.performance.performance_monitor import PerformanceMonitor
from rv_android_core.util.logging.manager import LoggingManager

# Integration example in action_service.py
class ActionService:
    def __init__(self):
        self.performance_monitor = PerformanceMonitor.get_instance()
        logging_manager = LoggingManager.get_instance()
        self.logger = logging_manager.get_logger(
            "rvandroid_tool.llm.service.action_service",
            {"component": "ActionService"}
        )
    
    def process_state(self, state: Dict[str, Any]) -> List[Dict[str, Any]]:
        with self.performance_monitor.measure_time("action_generation"):
            # Existing action generation logic
            generated_actions = self._generate_actions(state)
            
            # Log structured metrics for comparison
            self.logger.info(
                "Action generation completed",
                extra={
                    'metrics_type': 'action_generation',
                    'total_actions': len(generated_actions),
                    'mop_actions': sum(1 for a in generated_actions if getattr(a, 'directly_reaches_mop', False)),
                    'activity': state.get(StateEntry.ACTIVITY),
                    'strategy_type': self.strategy_type
                }
            )
            
            return generated_actions
```

## **Structured Logging for Performance Comparison**

### **Strategy Comparison Logging**
```python
# Example of structured logging for strategy comparison
def log_strategy_performance(self, strategy_name: str, metrics: Dict[str, Any]):
    """Log strategy performance with structured data."""
    
    self.logger.info(
        "Strategy performance metrics",
        extra={
            'metrics_type': 'strategy_performance',
            'strategy_name': strategy_name,
            'session_duration_minutes': metrics.get('session_duration', 0) / 60,
            'total_actions': metrics.get('total_actions', 0),
            'mop_violations_found': metrics.get('mop_violations', 0),
            'coverage_elements': metrics.get('coverage_elements', 0),
            'success_rate': metrics.get('success_rate', 0.0),
            'violations_per_hour': self._calculate_violations_per_hour(metrics)
        }
    )

def _calculate_violations_per_hour(self, metrics: Dict[str, Any]) -> float:
    """Calculate violations per hour for comparison."""
    duration_hours = metrics.get('session_duration', 0) / 3600
    violations = metrics.get('mop_violations', 0)
    return violations / duration_hours if duration_hours > 0 else 0.0
```

## **Usage Example**

```python
# Simple performance comparison using existing components
performance_monitor = PerformanceMonitor.get_instance()

# Baseline strategy run
with performance_monitor.measure_time("baseline_strategy"):
    baseline_results = run_baseline_strategy()

# Enhanced strategy run  
with performance_monitor.measure_time("enhanced_strategy"):
    enhanced_results = run_enhanced_strategy()

# Log comparison using structured logging
logger.info(
    "Strategy comparison completed",
    extra={
        'comparison_type': 'baseline_vs_enhanced',
        'baseline_violations': baseline_results['mop_violations'],
        'enhanced_violations': enhanced_results['mop_violations'],
        'improvement_factor': enhanced_results['mop_violations'] / max(1, baseline_results['mop_violations']),
        'baseline_coverage': baseline_results['coverage_elements'],
        'enhanced_coverage': enhanced_results['coverage_elements']
    }
)
```

This approach uses only existing system components without additional validation frameworks or complex comparison logic.
