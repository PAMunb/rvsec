## **Implementation Overview**

### **IMPORTANT DISTINCTION - Two Different Coverage Systems:**

#### **1. System Coverage Tracking (rv-coverage module)**
- **CoverageTracker**: Monitors logcat for method execution coverage from instrumented APK
- **Events**: `COVERAGE_UPDATED`, `MOP_ERROR_DETECTED` from CoverageTracker
- **Purpose**: Real-time monitoring of Java method execution and MOP violations
- **Data**: RvCoverageLog (method signatures), RvErrorLog (MOP violations)

#### **2. UI Coverage Tracking (UICoverageTracker - this plan)**  
- **UICoverageTracker**: Tracks which UI elements LLM has selected for testing
- **Purpose**: Systematic UI element exploration with [UNTESTED]/[TESTED] annotations
- **Data**: Element interaction counts and action type distribution
- **Different from**: System coverage - tracks LLM decisions, not APK execution

### **ActionService Integration**
- **System Coverage Events**: Subscribe to COVERAGE_UPDATED and MOP_ERROR_DETECTED from rv-coverage
- **UI Coverage Recording**: Record LLM action selections in UICoverageTracker  
- **Error Handling**: Handle both system coverage updates and action recording errors
- **Service Lifecycle**: Integration with rvandroid-tool server

```python
from rv_android_core.services.action_service import ActionService as BaseActionService
from rv_android_core.event.bus import EventBus
from rv_android_core.event.events import MOP_ERROR_DETECTED
from rv_android_core.domain.models import ActionResult, ActionType, ScreenItem
from rv_android_core.util.error.error_handler import ErrorHandler
from rv_android_core.constants import ACTION_SERVICE_CONSTANTS
from rvandroid.llm.constants import StateEntry

class ActionService(BaseActionService):
    """ActionService with UI coverage tracking and MOP event handling"""
    
    def __init__(self, ui_coverage_tracker: Optional['UICoverageTracker'] = None):
        super().__init__()
        self.ui_coverage_tracker = ui_coverage_tracker
        self.event_bus = EventBus.get_instance()
        self._setup_event_subscriptions()
        self.mop_error_handler = MOPErrorHandler(ui_coverage_tracker)
    
    def _setup_event_subscriptions(self):
        """Setup event subscriptions for system coverage and MOP events"""
        from rv_android_core.event.models import EventType
        
        # Subscribe to SYSTEM coverage updates from CoverageTracker (rv-coverage module)
        # This is DIFFERENT from UI Coverage Tracking (UICoverageTracker)
        self.event_bus.subscribe(
            EventType.COVERAGE_UPDATED,
            self._on_coverage_updated,
            filter_fn=lambda event: hasattr(event, 'source') and event.source == "CoverageTracker"
        )
        
        # Subscribe to MOP error detection from CoverageTracker 
        self.event_bus.subscribe(
            EventType.MOP_ERROR_DETECTED,
            self._on_mop_error_detected,
            filter_fn=lambda event: hasattr(event, 'source') and event.source == "CoverageTracker"  
        )
    
    @ErrorHandler.handle_errors(log_context="action_execution")
    def execute_action(self, action_data: Dict[str, Any]) -> ActionResult:
        """Execute action with coverage tracking"""
        
        # Extract action information
        action_type = ActionType(action_data.get('action_type'))
        screen_item = action_data.get('screen_item')
        
        # Execute the actual action
        result = super().execute_action(action_data)
        
        # Record simple interaction for coverage tracking
        if self.ui_coverage_tracker and screen_item:
            element_id = f"{screen_item.id}_{screen_item.text}"
            action_type_str = action_type.name.lower()
            
            self.ui_coverage_tracker.record_interaction(
                element_id=element_id,
                action_type=action_type_str
            )
        
        return result
    
    def _on_coverage_updated(self, event):
        """
        Handle SYSTEM coverage updates from CoverageTracker (rv-coverage module).
        
        This handles real-time method execution coverage updates from the instrumented APK,
        NOT UI element coverage from UICoverageTracker.
        """
        try:
            # Extract coverage metrics from CoverageTracker event
            coverage_data = event.data if hasattr(event, 'data') else {}
            
            # Store current system coverage metrics for context
            self.current_coverage_metrics = coverage_data
            
            self.logger.debug(f"System coverage updated: {coverage_data}")
            
        except Exception as e:
            self.logger.error(f"Error handling coverage update: {e}")
    
    def _on_mop_error_detected(self, event):
        """
        Handle MOP error detection from CoverageTracker (rv-coverage module).
        
        Stores recent MOP violations for testing context and guidance.
        """
        try:
            # Extract MOP error data
            error_data = event.data if hasattr(event, 'data') else {}
            
            # Add to recent errors list (keep last 5)
            self.recent_mop_errors.append(error_data)
            if len(self.recent_mop_errors) > 5:
                self.recent_mop_errors = self.recent_mop_errors[-5:]
            
            self.logger.info(f"MOP error detected: {error_data}")
            
        except Exception as e:
            self.logger.error(f"Error handling MOP error: {e}")
    
    @ErrorHandler.handle_errors(log_context="mop_error_handling")
    def _handle_mop_error_detected(self, event_data: Dict[str, Any]):
        """Handle MOP error detected events"""
        
        LoggingManager.info(
            "MOP violation detected",
            extra={
                'error_type': event_data.get('error_type'),
                'monitor_name': event_data.get('monitor_name'),
                'violation_context': event_data.get('context')
            }
        )
        
        # Record MOP violation in coverage tracker
        if self.ui_coverage_tracker:
            self.ui_coverage_tracker.record_mop_violation(
                error_type=event_data.get('error_type'),
                monitor_name=event_data.get('monitor_name'),
                context=event_data.get('context'),
                timestamp=datetime.utcnow()
            )
        
        # Let specialized handler process the violation
        self.mop_error_handler.process_violation(event_data)
    
    def _handle_action_executed(self, event_data: Dict[str, Any]):
        """Handle action executed events for pattern analysis"""
        
        if not self.ui_coverage_tracker:
            return
        
        # Update action type distribution
        action_type = event_data.get('action_type')
        if action_type:
            self.ui_coverage_tracker.update_action_distribution(action_type)
        
        # Check for successful navigation or state changes
        result = event_data.get('result')
        if result and result.success:
            screen_item = event_data.get('screen_item')
            if screen_item and (screen_item.reaches_mop or screen_item.directly_reaches_mop):
                self.ui_coverage_tracker.record_successful_mop_interaction(
                    screen_item=screen_item,
                    action_type=action_type,
                    result=result
                )
    
    def _handle_screen_transition(self, event_data: Dict[str, Any]):
        """Handle screen transition events"""
        
        if not self.ui_coverage_tracker:
            return
        
        # Record screen transition for navigation pattern analysis
        self.ui_coverage_tracker.record_screen_transition(
            from_screen=event_data.get('from_screen'),
            to_screen=event_data.get('to_screen'),
            trigger_action=event_data.get('trigger_action'),
            timestamp=datetime.utcnow()
        )
    
    def get_coverage_enhanced_context(self) -> Dict[str, Any]:
        """Get context data enhanced with coverage information"""
        
        base_context = super().get_context()
        
        if not self.ui_coverage_tracker:
            return base_context
        
        # Add coverage-specific context
        coverage_context = {
            'coverage_statistics': self.ui_coverage_tracker.get_coverage_statistics(),
            'action_distribution': self.ui_coverage_tracker.get_action_type_distribution(),
            'recent_mop_interactions': self.ui_coverage_tracker.get_recent_mop_interactions(limit=5),
            'successful_patterns': self.ui_coverage_tracker.get_successful_patterns(),
            'untested_mop_elements': self.ui_coverage_tracker.get_untested_mop_elements_count()
        }
        
        return {**base_context, **coverage_context}
```

```python
class UICoverageActionHandler:
    """Example handler showing proper StateEntry usage with UICoverageTracker"""
    
    def __init__(self, ui_coverage_tracker: 'UICoverageTracker'):
        self.ui_coverage_tracker = ui_coverage_tracker
        self.logger = LoggingManager.get_instance().get_logger(
            "rvandroid_tool.coverage.action_handler",
            {"component": "UICoverageActionHandler"}
        )
    
    def handle_action_execution(self, state: Dict[str, Any], action: Any) -> None:
        """
        Handle action execution with proper StateEntry usage.
        
        Args:
            state: Current application state dictionary
            action: Action being executed with item property
        """
        try:
            # Proper usage of StateEntry constants
            element_id = f"{state.get(StateEntry.ACTIVITY)}_{action.item.id}"
            
            # Record the action in coverage tracker
            self.ui_coverage_tracker.record_interaction(
                element_id=element_id,
                screen_hash=state.get(StateEntry.HASH_SCREEN_CONTENT),
                activity=state.get(StateEntry.ACTIVITY),
                action_type=action.action_type
            )
            
            # Add coverage annotation to action
            coverage_info = self.ui_coverage_tracker.get_element_coverage_info(element_id)
            action.coverage_annotation = f" [TESTED:{coverage_info.interaction_count}x]"
            
            self.logger.debug(
                f"Recorded action execution: element_id={element_id}, "
                f"activity={state.get(StateEntry.ACTIVITY)}, "
                f"interactions={coverage_info.interaction_count}"
            )
            
        except Exception as e:
            self.logger.error(f"Error handling action execution: {e}")
    
    def enrich_state_with_coverage(self, state: Dict[str, Any]) -> Dict[str, Any]:
        """
        Enrich state with coverage information using proper StateEntry constants.
        
        Args:
            state: Current application state
            
        Returns:
            Enhanced state with coverage data
        """
        try:
            # Get current screen context using StateEntry constants
            current_activity = state.get(StateEntry.ACTIVITY)
            screen_hash = state.get(StateEntry.HASH_SCREEN_CONTENT)
            structured_screen = state.get(StateEntry.STRUCTURED_SCREEN)
            
            if not (current_activity and screen_hash and structured_screen):
                return state
            
            # Add coverage statistics to state
            coverage_stats = self.ui_coverage_tracker.get_coverage_statistics()
            state['coverage_metrics'] = {
                'total_elements': coverage_stats.total_elements,
                'tested_elements': coverage_stats.tested_elements,
                'coverage_rate': coverage_stats.coverage_rate,
                'current_activity': current_activity,
                'screen_hash': screen_hash
            }
            
            # Annotate screen elements with coverage info
            if hasattr(structured_screen, 'items'):
                for item in structured_screen.items:
                    for action in item.actions:
                        element_id = f"{current_activity}_{action.id}"
                        coverage_info = self.ui_coverage_tracker.get_element_coverage_info(element_id)
                        
                        if coverage_info:
                            action.coverage_annotation = f" [TESTED:{coverage_info.interaction_count}x]"
                        else:
                            action.coverage_annotation = " [UNTESTED]"
            
            self.logger.debug(f"Enhanced state with coverage data for activity: {current_activity}")
            return state
            
        except Exception as e:
            self.logger.error(f"Error enriching state with coverage: {e}")
            return state
```

```python
"""
Cache system for ActionService and UICoverageTracker performance.

Provides caching with timeout-based expiration to improve performance.
"""

import time
from typing import Dict, Any, Optional

from rv_android_core.util.logging.manager import LoggingManager


class Cache:
    """
    Dictionary-based cache with timeout expiration.
    
    Provides caching functionality with automatic cleanup of expired
    entries to improve performance for ActionService and UICoverageTracker operations.
    """
    
    def __init__(self, default_timeout: float = 300.0):  # 5 minute default
        self.cache: Dict[str, tuple] = {}  # key -> (value, expiry_time)
        self.default_timeout = default_timeout
        
        # Set up logging
        logging_manager = LoggingManager.get_instance()
        self.logger = logging_manager.get_logger(
            "rvandroid_tool.core.memory.simple_cache",
            {"component": "Cache"}
        )
    
    def get(self, key: str) -> Optional[Any]:
        """Get value from cache if not expired."""
        if key not in self.cache:
            return None
            
        value, expiry_time = self.cache[key]
        
        if time.time() > expiry_time:
            # Expired - remove and return None
            del self.cache[key]
            return None
            
        return value
    
    def set(self, key: str, value: Any, timeout: Optional[float] = None) -> None:
        """Set value in cache with timeout."""
        if timeout is None:
            timeout = self.default_timeout
            
        expiry_time = time.time() + timeout
        self.cache[key] = (value, expiry_time)
    
    def invalidate(self, key: str) -> bool:
        """Remove specific key from cache."""
        if key in self.cache:
            del self.cache[key]
            return True
        return False
    
    def invalidate_pattern(self, pattern: str) -> int:
        """Remove all keys containing pattern. Returns count removed."""
        count = 0
        keys_to_remove = []
        
        for key in self.cache.keys():
            if pattern in key:
                keys_to_remove.append(key)
        
        for key in keys_to_remove:
            del self.cache[key]
            count += 1
            
        return count
    
    def clear(self) -> None:
        """Clear all cache entries."""
        self.cache.clear()
        
    def cleanup_expired(self) -> int:
        """Remove expired entries. Returns count removed."""
        current_time = time.time()
        expired_keys = []
        
        for key, (value, expiry_time) in self.cache.items():
            if current_time > expiry_time:
                expired_keys.append(key)
        
        for key in expired_keys:
            del self.cache[key]
            
        return len(expired_keys)
    
    def size(self) -> int:
        """Return current cache size."""
        return len(self.cache)


# Global cache instances for different components
action_service_cache = Cache(timeout=180.0)  # 3 minutes for action context
ui_coverage_cache = Cache(timeout=300.0)     # 5 minutes for coverage data
```

```python
class CoverageErrorRecovery:
    """Error recovery and graceful degradation for coverage features"""
    
    @staticmethod
    @ErrorHandler.handle_errors(default_return=None, log_context="coverage_recovery")
    def safe_coverage_operation(operation: Callable, fallback: Callable = None) -> Any:
        """Safely execute coverage operations with fallback"""
        
        try:
            return operation()
        except Exception as e:
            LoggingManager.warning(
                f"Coverage operation failed, using fallback: {e}",
                exception=e
            )
            
            if fallback:
                return fallback()
            
            return None
    
    @staticmethod
    def create_minimal_coverage_context() -> Dict[str, Any]:
        """Create minimal coverage context when full system fails"""
        
        return {
            'coverage_enabled': False,
            'show_coverage_stats': False,
            'coverage_guidance_enabled': False,
            'action_balance_enabled': False,
            'ui_coverage_tracker': None
        }
    
    @staticmethod
    def validate_coverage_integration() -> bool:
        """Validate coverage system integration"""
        
        try:
            # Test key components
            event_bus = EventBus.get_instance()
            service_manager = ServiceManager.get_instance()
            
            # Verify event system
            if not hasattr(event_bus, 'publish'):
                return False
            
            # Verify service integration
            if not service_manager:
                return False
            
            return True
            
        except Exception as e:
            LoggingManager.error(f"Coverage integration validation failed: {e}")
            return False
```
