class UIElementsFragment(BaseFragment):
    """Enhanced UIElementsFragment with universal coverage tracking"""
    
    def __init__(self, ui_coverage_tracker: Optional['UICoverageTracker'] = None):
        super().__init__()
        self.ui_coverage_tracker = ui_coverage_tracker
        self.required_variables = [StateEntry.SCREEN_DESCRIPTION]
        self.optional_variables = ['coverage_enabled', 'show_coverage_stats']
    
    def generate_content(self, context: Dict[str, Any]) -> str:
        """Generate UI elements content with coverage annotations"""
        screen_description: ScreenDescription = context[StateEntry.SCREEN_DESCRIPTION]
        coverage_enabled = context.get('coverage_enabled', True)
        
        if not coverage_enabled or not self.ui_coverage_tracker:
            return str(screen_description)
        
        # Apply coverage annotations to screen description
        annotated_description = self._apply_coverage_annotations(screen_description)
        
        # Add coverage statistics if requested
        if context.get('show_coverage_stats', False):
            stats_section = self._generate_coverage_stats()
            return f"{annotated_description}\n\n{stats_section}"
        
        return annotated_description
    
    def _apply_coverage_annotations(self, screen_description: ScreenDescription) -> str:
        """Apply coverage annotations to ScreenItem elements"""
        original_str = str(screen_description)
        
        for item in screen_description.items:
            if hasattr(item, 'text') and item.text:
                coverage_info = self.ui_coverage_tracker.get_element_coverage(item)
                annotation = self._create_coverage_annotation(coverage_info)
                
                if annotation:
                    # Replace original item text with annotated version
                    annotated_text = f"{item.text} {annotation}"
                    original_str = original_str.replace(item.text, annotated_text, 1)
        
        return original_str
    
    def _create_coverage_annotation(self, coverage_info: ElementCoverage) -> str:
        """Create coverage annotation based on element interaction history"""
        if coverage_info.selection_count == 0:
            return "[UNTESTED]"
        elif coverage_info.selection_count == 1:
            return "[TESTED-1x]"
        elif coverage_info.selection_count <= 3:
            return f"[TESTED-{coverage_info.selection_count}x]"
        else:
            return "[WELL-TESTED]"
    
    def _generate_coverage_stats(self) -> str:
        """Generate coverage statistics section"""
        stats = self.ui_coverage_tracker.get_coverage_statistics()
        
        return f"""## UI Coverage Statistics
**Total Elements**: {stats.total_elements}
**Tested Elements**: {stats.tested_elements}
**Coverage Rate**: {stats.coverage_rate:.1%}
**Action Distribution**: Click({stats.action_distribution.click}), SetText({stats.action_distribution.set_text}), Coordinate({stats.action_distribution.coordinate})"""
```

from rv_android_core.domain.models import ActionType
from rv_android_core.constants import UI_COVERAGE_CONSTANTS

class CoverageGuidanceFragment(BaseFragment):
    """Provides LLM guidance for balanced UI exploration"""
    
    def __init__(self, ui_coverage_tracker: Optional['UICoverageTracker'] = None):
        super().__init__()
        self.ui_coverage_tracker = ui_coverage_tracker
        self.required_variables = []
        self.optional_variables = ['coverage_guidance_enabled', 'detailed_guidance']
    
    def generate_content(self, context: Dict[str, Any]) -> str:
        """Generate coverage guidance content for LLM"""
        if not context.get('coverage_guidance_enabled', True):
            return ""
        
        if not self.ui_coverage_tracker:
            return self._generate_basic_guidance()
        
        guidance_sections = []
        
        # Element coverage guidance
        untested_guidance = self._generate_untested_elements_guidance()
        if untested_guidance:
            guidance_sections.append(untested_guidance)
        
        # Action type balance guidance
        balance_guidance = self._generate_action_balance_guidance()
        if balance_guidance:
            guidance_sections.append(balance_guidance)
        
        # Screen pattern specific guidance
        pattern_guidance = self._generate_pattern_specific_guidance(context)
        if pattern_guidance:
            guidance_sections.append(pattern_guidance)
        
        if not guidance_sections:
            return ""
        
        return f"## Exploration Guidance\n\n" + "\n\n".join(guidance_sections)
    
    def _generate_untested_elements_guidance(self) -> str:
        """Generate guidance for untested elements"""
        stats = self.ui_coverage_tracker.get_coverage_statistics()
        
        if stats.untested_elements == 0:
            return "✓ **Coverage Status**: All visible elements have been tested"
        
        untested_count = stats.untested_elements
        total_count = stats.total_elements
        
        guidance = f"📊 **Element Coverage**: {untested_count}/{total_count} elements remain [UNTESTED]"
        
        if untested_count > 5:
            guidance += " - **Priority**: Focus on systematic element exploration"
        elif untested_count <= 3:
            guidance += " - **Priority**: Complete remaining element coverage"
        
        return guidance
    
    def _generate_action_balance_guidance(self) -> str:
        """Generate action type balance recommendations"""
        distribution = self.ui_coverage_tracker.get_action_type_distribution()
        total_actions = sum(distribution.values())
        
        if total_actions < UI_COVERAGE_CONSTANTS.MIN_ACTIONS_FOR_BALANCE_ANALYSIS:
            return ""
        
        # Calculate percentages
        percentages = {
            action_type: (count / total_actions) * 100 
            for action_type, count in distribution.items()
        }
        
        recommendations = []
        
        # Detect imbalances and provide recommendations
        if percentages.get(ActionType.CLICK, 0) > 70:
            recommendations.append("⚖️ **Action Balance**: Consider more text input and coordinate actions")
        
        if percentages.get(ActionType.SET_TEXT, 0) < 10 and self._has_input_elements():
            recommendations.append("📝 **Text Input**: Explore text input fields for comprehensive testing")
        
        if percentages.get(ActionType.COORDINATE, 0) < 5:
            recommendations.append("🎯 **Coordinate Actions**: Consider custom coordinate interactions for non-standard elements")
        
        if not recommendations:
            return "✓ **Action Balance**: Good distribution of action types"
        
        return "\n".join(recommendations)
    
    def _generate_pattern_specific_guidance(self, context: Dict[str, Any]) -> str:
        """Generate pattern-specific guidance based on screen context"""
        screen_description = context.get('screen_description')
        if not screen_description:
            return ""
        
        # Detect screen patterns and provide specific guidance
        screen_text = str(screen_description).lower()
        
        guidance_items = []
        
        # Form pattern guidance
        if any(keyword in screen_text for keyword in ['form', 'input', 'field', 'submit', 'save']):
            guidance_items.append("📋 **Form Pattern**: Complete input fields before triggering submit/validation actions")
        
        # Navigation pattern guidance
        if any(keyword in screen_text for keyword in ['menu', 'tab', 'navigation', 'drawer']):
            guidance_items.append("🧭 **Navigation Pattern**: Explore menu/tabs systematically for activity coverage")
        
        # Settings/Config pattern guidance
        if any(keyword in screen_text for keyword in ['settings', 'preferences', 'config', 'options']):
            guidance_items.append("⚙️ **Configuration Pattern**: Test setting changes for validation/permission triggers")
        
        # High MOP density guidance
        mop_count = screen_text.count('[M]') + screen_text.count('[DM]')
        if mop_count > 3:
            guidance_items.append(f"🎯 **High MOP Density**: {mop_count} monitored operations available - test systematically")
        
        return "\n".join(guidance_items) if guidance_items else ""
    
    def _has_input_elements(self) -> bool:
        """Check if current screen has input elements that haven't been tested"""
        # Implementation would check for EditText, Spinner, etc. elements
        return True  # Simplified for plan
    
    def _generate_basic_guidance(self) -> str:
        """Generate basic guidance when coverage tracker is unavailable"""
        return """## Exploration Guidance

📊 **Systematic Exploration**: Focus on testing all visible UI elements systematically
⚖️ **Action Balance**: Maintain good distribution of click, text input, and coordinate actions
🎯 **MOP Priority**: Prioritize [M] and [DM] elements while considering screen context"""

from rv_android_core.constants import MOP_CONSTANTS

class MonitoredOperationsFragment(BaseFragment):
    """Enhanced monitored operations fragment with context intelligence"""
    
    def __init__(self, ui_coverage_tracker: Optional['UICoverageTracker'] = None):
        super().__init__()
        self.ui_coverage_tracker = ui_coverage_tracker
        self.required_variables = [StateEntry.MONITORED_OPERATIONS]
        self.optional_variables = ['mop_context_analysis', 'mop_history_enabled']
    
    def generate_content(self, context: Dict[str, Any]) -> str:
        """Generate enhanced MOP content with context analysis"""
        mop_data = context[StateEntry.MONITORED_OPERATIONS]
        
        content_sections = []
        
        # Basic MOP information
        basic_info = self._generate_basic_mop_info(mop_data)
        content_sections.append(basic_info)
        
        # Context analysis if enabled
        if context.get('mop_context_analysis', True):
            context_analysis = self._generate_mop_context_analysis(context)
            if context_analysis:
                content_sections.append(context_analysis)
        
        # MOP history analysis if available
        if context.get('mop_history_enabled', True) and self.ui_coverage_tracker:
            history_analysis = self._generate_mop_history_analysis()
            if history_analysis:
                content_sections.append(history_analysis)
        
        return "\n\n".join(content_sections)
    
    def _generate_basic_mop_info(self, mop_data: Dict[str, Any]) -> str:
        """Generate basic monitored operations information"""
        coverage_info = mop_data.get('coverage', {})
        recent_errors = mop_data.get('recent_errors', [])
        
        content = f"""## Monitored Operations Status

**Coverage Progress**: {coverage_info.get('covered_operations', 0)}/{coverage_info.get('total_operations', 0)} operations
**Recent MOP Errors**: {len(recent_errors)} violations detected"""
        
        if recent_errors:
            content += f"\n**Latest Error**: {recent_errors[-1].get('error_type', 'Unknown')}"
        
        return content
    
    def _generate_mop_context_analysis(self, context: Dict[str, Any]) -> str:
        """Generate context-aware MOP analysis"""
        screen_description = context.get('screen_description')
        if not screen_description:
            return ""
        
        analysis_items = []
        
        # Analyze M/DM distribution and context
        m_count = str(screen_description).count('[M]')
        dm_count = str(screen_description).count('[DM]')
        
        if dm_count > 0:
            analysis_items.append(f"🎯 **Direct MOP Access**: {dm_count} [DM] elements provide direct monitor access")
        
        if m_count > dm_count:
            analysis_items.append(f"🔄 **Indirect MOP Access**: {m_count} [M] elements require navigation sequences")
        
        # Screen pattern analysis for MOP strategy
        screen_text = str(screen_description).lower()
        
        if 'form' in screen_text and dm_count > 0:
            analysis_items.append("📋 **Form Strategy**: Complete input fields before triggering [DM] submit/validate actions")
        
        if any(nav in screen_text for nav in ['tab', 'menu', 'navigation']) and m_count > 0:
            analysis_items.append("🧭 **Navigation Strategy**: Explore tabs/menus to discover additional MOP surfaces")
        
        if dm_count > 2:
            analysis_items.append("🔥 **High Priority Screen**: Multiple direct MOP access points - systematic testing recommended")
        
        if not analysis_items:
            return ""
        
        return f"## MOP Context Analysis\n\n" + "\n".join(analysis_items)
    
    def _generate_mop_history_analysis(self) -> str:
        """Generate MOP history and pattern analysis"""
        if not self.ui_coverage_tracker:
            return ""
        
        # Get MOP-related interaction history
        mop_interactions = self.ui_coverage_tracker.get_mop_interactions_history()
        
        if not mop_interactions:
            return ""
        
        analysis_items = []
        
        # Successful violation patterns
        successful_violations = [
            interaction for interaction in mop_interactions 
            if interaction.resulted_in_mop_error
        ]
        
        if successful_violations:
            analysis_items.append(f"✅ **Successful Patterns**: {len(successful_violations)} MOP violations discovered through systematic testing")
            
            # Pattern analysis
            successful_actions = [v.action_type for v in successful_violations]
            if successful_actions.count(ActionType.CLICK) > len(successful_actions) * 0.6:
                analysis_items.append("🎯 **Pattern Insight**: Click actions showing high MOP violation success rate")
        
        # Untested MOP elements
        untested_mop_count = self.ui_coverage_tracker.get_untested_mop_elements_count()
        if untested_mop_count > 0:
            analysis_items.append(f"🔍 **Exploration Opportunity**: {untested_mop_count} [M]/[DM] elements remain untested")
        
        if not analysis_items:
            return ""
        
        return f"## MOP Pattern Analysis\n\n" + "\n".join(analysis_items)

from rv_android_core.domain.factory import ComponentFactory

class FragmentFactory(ComponentFactory):
    """Enhanced fragment factory with coverage support"""
    
    def __init__(self):
        super().__init__()
        self._coverage_aware_fragments = {
            'ui_elements',
            'coverage_guidance', 
            'monitored_operations_enhanced'
        }
    
    def create_fragment(self, fragment_name: str, context: Dict[str, Any]) -> BaseFragment:
        """Create fragment with coverage support if available"""
        
        # Check if fragment supports coverage
        if fragment_name in self._coverage_aware_fragments:
            ui_coverage_tracker = context.get('ui_coverage_tracker')
            
            # Enhanced fragments with coverage support
            if fragment_name == 'ui_elements':
                return UIElementsFragment(ui_coverage_tracker=ui_coverage_tracker)
            elif fragment_name == 'coverage_guidance':
                return CoverageGuidanceFragment(ui_coverage_tracker=ui_coverage_tracker)
            elif fragment_name == 'monitored_operations_enhanced':
                return MonitoredOperationsFragment(ui_coverage_tracker=ui_coverage_tracker)
        
        # Fall back to standard fragment creation
        return super().create_fragment(fragment_name, context)
    
    def register_enhanced_fragments(self):
        """Register all enhanced fragments"""
        enhanced_fragments = {
            'ui_elements': UIElementsFragment,
            'coverage_guidance': CoverageGuidanceFragment,
            'monitored_operations_enhanced': MonitoredOperationsFragment,
        }
        
        for name, fragment_class in enhanced_fragments.items():
            self.register_fragment(name, fragment_class)
```

"""
Centralized screen pattern analysis for reusable pattern detection.

This module provides centralized screen pattern detection capabilities that can
be used across different strategies and fragments. Based on analysis feedback
to centralize pattern detection logic previously scattered across components.
"""

from typing import Dict, List, Set, Any, Optional, Tuple
from dataclasses import dataclass
from enum import Enum

from rv_screen_parser.parser.screen.visitor.model import ScreenDescription, ItemAction
from rv_android_core.constants import UI_COVERAGE_CONSTANTS

class ScreenPatternType(Enum):
    """Screen pattern classification types"""
    FORM_HEAVY = "form_heavy"
    NAVIGATION_HEAVY = "navigation_heavy"
    LIST_HEAVY = "list_heavy"
    DIALOG = "dialog"
    MINIMAL = "minimal"
    CONFIGURATION = "configuration"
    HIGH_MOP_DENSITY = "high_mop_density"
    STANDARD = "standard"

@dataclass
class PatternDetectionResult:
    """Result of pattern detection analysis"""
    
    pattern_type: ScreenPatternType
    confidence_score: float
    supporting_evidence: List[str]
    element_classifications: Dict[str, str]

@dataclass
class ScreenComplexityMetrics:
    """Comprehensive screen complexity analysis"""
    
    total_elements: int
    clickable_elements: int
    input_elements: int
    navigation_elements: int
    form_elements: int
    mop_elements: int
    
    # Calculated metrics
    clickable_ratio: float
    input_density: float
    mop_density: float
    complexity_score: float
    interaction_potential: float

@dataclass
class ScreenPatternResult:
    """Complete screen pattern analysis result"""
    
    screen_hash: str
    pattern_detection: PatternDetectionResult
    complexity_metrics: ScreenComplexityMetrics
    form_elements: List[Dict[str, Any]]
    navigation_elements: List[Dict[str, Any]]
    mop_elements: List[Dict[str, Any]]
    interaction_areas: Dict[str, int]
    
    # Strategy recommendations
    recommended_strategy: str
    strategy_confidence: float
    testing_focus_areas: List[str]

class ScreenPatternAnalyzer:
    """
    Minimal screen pattern detection for testing guidance.
    """
    
    def __init__(self):
        pass  # No state needed for simple analyzer
    
    def analyze_screen(self, screen_description) -> str:
        """
        Detect basic screen pattern for testing guidance.
        
        Returns: 'form', 'list', 'navigation', or 'content'
        """
        if not hasattr(screen_description, 'items'):
            return 'content'
        
        # Count element types
        input_count = 0
        button_count = 0
        has_list = False
        
        for item in screen_description.items:
            widget_type = getattr(item, 'widget_type', '').lower()
            
            if 'edit' in widget_type or 'input' in widget_type:
                input_count += 1
            if 'button' in widget_type:
                button_count += 1
            if 'list' in widget_type or 'recycler' in widget_type:
                has_list = True
        
        # Simple classification
        if input_count >= 2 and button_count >= 1:
            return 'form'
        elif has_list:
            return 'list'
        elif button_count >= 3:
            return 'navigation'
        else:
            return 'content'
    
    def get_testing_focus(self, pattern: str) -> str:
        """Get simple testing focus based on pattern."""
        focus_map = {
            'form': "Fill all input fields before submitting",
            'list': "Explore list items systematically",
            'navigation': "Test all navigation options",
            'content': "General exploration of interactive elements"
        }
        return focus_map.get(pattern, "General exploration")

# Usage in MOPVisionStrategy or fragments  
pattern_analyzer = ScreenPatternAnalyzer()
pattern = pattern_analyzer.analyze_screen(screen_description)
focus = pattern_analyzer.get_testing_focus(pattern)
