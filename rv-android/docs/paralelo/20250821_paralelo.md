### 1.2 Fluxo de Execução Atual e Identificação de Pontos de Paralelização

**Fluxo de Execução Atual:**
1. **rv-experiment** (orquestrador): Coordena todo o processo
2. **Pre-processamento sequencial**: 
   - `rv-monitor-generator`: Gera monitores MOP → `monitors/` (uma vez)
   - `rv-instrumentation`: Instrumenta todos os APKs → `instrumented/` (uma vez)
   - `rv-static-analysis`: Analisa APKs originais → `static_analysis/` (uma vez)
3. **rv-platform**: Executa tasks sequencialmente nos APKs
4. **Post-processamento**: Agrega resultados finais

**Análise Crítica para Paralelização:**

**PortManager - Gerenciamento de Portas:**
```python
# modules/rv-android-core/src/rv_android_core/util/port_manager.py
from typing import Set, Optional
import threading
from rv_android_core.constants import MIN_EMULATOR_PORT, MAX_EMULATOR_PORT, MIN_SERVER_PORT, MAX_SERVER_PORT

class PortManager:
    """Gerenciador de alocação de portas para isolamento de processos"""
    
    _allocated_emulator_ports: Set[int] = set()
    _allocated_server_ports: Set[int] = set()
    _lock = threading.Lock()
    
    @classmethod
    def assign_emulator_port(cls) -> int:
        """Assign unique emulator port for process"""
        with cls._lock:
            for port in range(MIN_EMULATOR_PORT, MAX_EMULATOR_PORT + 1, 2):  # ADB uses port pairs
                if port not in cls._allocated_emulator_ports:
                    cls._allocated_emulator_ports.add(port)
                    return port
            raise ResourceAllocationError("No available emulator ports")
    
    @classmethod 
    def assign_server_port(cls) -> int:
        """Assign unique server port for process"""
        with cls._lock:
            for port in range(MIN_SERVER_PORT, MAX_SERVER_PORT + 1):
                if port not in cls._allocated_server_ports:
                    cls._allocated_server_ports.add(port)
                    return port
            raise ResourceAllocationError("No available server ports")
    
    @classmethod
    def release_ports(cls, emulator_port: int, server_port: int):
        """Release allocated ports"""
        with cls._lock:
            cls._allocated_emulator_ports.discard(emulator_port)
            cls._allocated_server_ports.discard(server_port)
```

### 6.1 PortManager - Alocação Dinâmica de Portas

**Responsabilidade:** Gerenciar pool de portas para evitar conflitos

```python
# modules/rv-android-core/src/rv_android_core/util/resources/port_manager.py
class PortManager:
    """Gerenciador de portas para isolamento de processos"""
    
    # Pool de portas para emuladores (pares para ADB)
    _EMULATOR_PORT_POOL = list(range(5554, 5594, 2))  # 20 emuladores simultâneos
    
    # Pool de portas para servidores rvandroid-tool
    _SERVER_PORT_POOL = list(range(5000, 5020))       # 20 servidores simultâneos
    
    _assigned_ports: Set[int] = set()
    _port_owners: Dict[int, str] = {}  # port -> process_id
    _lock = threading.Lock()
    
    @classmethod
    def assign_emulator_port(cls, process_id: str = None) -> int:
        """Atribui porta única para emulador"""
        with cls._lock:
            for port in cls._EMULATOR_PORT_POOL:
                if port not in cls._assigned_ports:
                    cls._assigned_ports.add(port)
                    if process_id:
                        cls._port_owners[port] = process_id
                    return port
            raise ResourceAllocationError("No available emulator ports")
    
    @classmethod  
    def assign_server_port(cls, process_id: str = None) -> int:
        """Atribui porta única para servidor"""
        with cls._lock:
            for port in cls._SERVER_PORT_POOL:
                if port not in cls._assigned_ports:
                    cls._assigned_ports.add(port)
                    if process_id:
                        cls._port_owners[port] = process_id
                    return port
            raise ResourceAllocationError("No available server ports")
    
    @classmethod
    def release_port(cls, port: int) -> bool:
        """Libera porta para reuso"""
        with cls._lock:
            if port in cls._assigned_ports:
                cls._assigned_ports.remove(port)
                cls._port_owners.pop(port, None)
                return True
            return False
    
    @classmethod
    def release_process_ports(cls, process_id: str) -> int:
        """Libera todas as portas de um processo"""
        with cls._lock:
            released = 0
            to_release = [port for port, owner in cls._port_owners.items() if owner == process_id]
            for port in to_release:
                cls._assigned_ports.discard(port)
                cls._port_owners.pop(port, None)
                released += 1
            return released
    
    @classmethod
    def get_port_status(cls) -> Dict[str, Any]:
        """Retorna status de utilização de portas"""
        with cls._lock:
            return {
                'emulator_ports_used': len([p for p in cls._assigned_ports if p in cls._EMULATOR_PORT_POOL]),
                'emulator_ports_available': len([p for p in cls._EMULATOR_PORT_POOL if p not in cls._assigned_ports]),
                'server_ports_used': len([p for p in cls._assigned_ports if p in cls._SERVER_PORT_POOL]),
                'server_ports_available': len([p for p in cls._SERVER_PORT_POOL if p not in cls._assigned_ports]),
                'assigned_ports': list(cls._assigned_ports),
                'port_owners': cls._port_owners.copy()
            }
```

```python
# modules/rv-platform/src/rv_platform/execution/parallel_executor.py
class ParallelTaskExecutor:
    """Executor de tasks em processos paralelos isolados"""
    
    def __init__(self, platform_config: PlatformConfig):
        self.platform_config = platform_config
        self.max_processes = min(platform_config.max_parallel_processes, 
                               self._calculate_optimal_processes())
        
        # Setup logging
        self.logger = LoggingManager.get_instance().get_logger(
            'rv_platform.execution.parallel_executor',
            {CONTEXT_COMPONENT: 'ParallelTaskExecutor'}
        )
        
        # Resource managers  
        self.process_pool = None
        self.active_processes: Dict[str, Dict] = {}
        
    def execute_tasks_parallel(self, tasks: List[Task]) -> Dict[str, TaskResult]:
        """
        Executa tasks em processos paralelos isolados
        
        Args:
            tasks: Lista de tasks para executar
            
        Returns:
            Dict mapeando task_id -> TaskResult
        """
        if not tasks:
            return {}
            
        # Check if parallel execution is viable
        if not self._can_execute_parallel(tasks):
            return self._execute_sequential_fallback(tasks)
            
        results = {}
        
        try:
            # Initialize process pool
            self.process_pool = ProcessPoolExecutor(max_workers=self.max_processes)
            
            # Create process configurations (usando artefatos compartilhados)
            shared_config = SharedArtifactsConfig(self.platform_config.base_output_dir)
            process_configs = self._create_process_configurations(tasks, shared_config)
            
            # Submit tasks to processes
            future_to_task = {}
            for task, process_config in zip(tasks, process_configs):
                future = self.process_pool.submit(
                    execute_task_in_isolated_process,
                    task.to_dict(),
                    process_config.to_dict(),
                    self.platform_config.to_dict()
                )
                future_to_task[future] = task
                self.active_processes[task.id] = {
                    'process_config': process_config,
                    'future': future,
                    'start_time': time.time()
                }
            
            # Collect results as they complete
            for future in concurrent.futures.as_completed(future_to_task, timeout=7200):  # 2h timeout
                task = future_to_task[future]
                try:
                    result = future.result()
                    results[task.id] = TaskResult.from_dict(result)
                    self.logger.info(f"Task {task.id} completed successfully")
                    
                    # Cleanup process resources
                    self._cleanup_process_resources(task.id)
                    
                except Exception as e:
                    self.logger.error(f"Task {task.id} failed: {e}")
                    results[task.id] = TaskResult.create_error(task.id, str(e))
                    
        except Exception as e:
            self.logger.error(f"Parallel execution failed: {e}")
            # Force cleanup zombie processes before fallback
            self._force_cleanup_zombie_processes()
            self._cleanup_all_processes()
            return self._execute_sequential_fallback(tasks)
            
        finally:
            # Robust shutdown sequence
            if self.process_pool:
                try:
                    # Try graceful shutdown first
                    self.process_pool.shutdown(wait=True, timeout=30)
                except Exception as shutdown_error:
                    self.logger.warning(f"Error during pool shutdown: {shutdown_error}")
                    # Force shutdown if graceful fails
                    self.process_pool.shutdown(wait=False)
            
            # Final cleanup of any remaining resources
            self._force_cleanup_zombie_processes()
                
        return results
    
    def _create_process_configurations(self, tasks: List[Task], shared_config: SharedArtifactsConfig) -> List[ProcessConfiguration]:
        """Cria configurações isoladas para cada processo usando artefatos compartilhados"""
        configs = []
        for i, task in enumerate(tasks):
            process_id = f"{task.id}_{int(time.time())}_{i}"
            config = ProcessConfiguration(process_id, shared_config, self.platform_config.to_dict())
            configs.append(config)
        return configs
    
    def _can_execute_parallel(self, tasks: List[Task]) -> bool:
        """Verifica se execução paralela é viável"""
        # Check resource availability
        port_status = PortManager.get_port_status()
        available_emulator_ports = port_status['emulator_ports_available']
        available_server_ports = port_status['server_ports_available']
        
        required_processes = min(len(tasks), self.max_processes)
        
        if available_emulator_ports < required_processes:
            self.logger.warning(f"Insufficient emulator ports: need {required_processes}, have {available_emulator_ports}")
            return False
            
        if available_server_ports < required_processes:
            self.logger.warning(f"Insufficient server ports: need {required_processes}, have {available_server_ports}")
            return False
            
        # Check system resources (CPU, memory)
        if not self._check_system_resources(required_processes):
            return False
            
        return True
    
    def _cleanup_process_resources(self, task_id: str):
        """Clean up resources for a specific process with robust error handling"""
        if task_id in self.active_processes:
            process_config = self.active_processes[task_id]['process_config']
            
            try:
                # Release ports
                released_ports = PortManager.release_process_ports(process_config.process_id)
                self.logger.debug(f"Released {released_ports} ports for process {process_config.process_id}")
                
                # Cleanup directories if configured
                if self.platform_config.cleanup_process_dirs:
                    PathManager.cleanup_process_directories(process_config.process_id)
                    
            except Exception as e:
                self.logger.warning(f"Error during resource cleanup for {task_id}: {e}")
                
            finally:
                # Always remove from active processes
                del self.active_processes[task_id]

    def _force_cleanup_zombie_processes(self):
        """Force cleanup of zombie processes and their resources - CRITICAL for robustness"""
        zombie_tasks = []
        
        for task_id, process_info in self.active_processes.items():
            try:
                future = process_info['future']
                start_time = process_info['start_time']
                
                # Check if process is taking too long (zombie detection)
                if time.time() - start_time > self.platform_config.process_timeout:
                    self.logger.warning(f"Process timeout exceeded for task {task_id}, forcing cleanup")
                    
                    # Cancel future if not done
                    if not future.done():
                        future.cancel()
                        
                    zombie_tasks.append(task_id)
                    
            except Exception as e:
                self.logger.error(f"Error checking process status for {task_id}: {e}")
                zombie_tasks.append(task_id)
        
        # Clean up zombie processes
        for task_id in zombie_tasks:
            try:
                self._cleanup_process_resources(task_id)
                self.logger.warning(f"Force cleaned up zombie process for task {task_id}")
            except Exception as e:
                self.logger.error(f"Error during force cleanup of {task_id}: {e}")
    
    def _cleanup_all_processes(self):
        """Clean up all active processes - called during shutdown"""
        self.logger.info("Cleaning up all active processes")
        
        # First try graceful cleanup
        active_task_ids = list(self.active_processes.keys())
        for task_id in active_task_ids:
            self._cleanup_process_resources(task_id)
        
        # Force cleanup any remaining zombies
        if self.active_processes:
            self.logger.warning(f"Force cleaning up {len(self.active_processes)} remaining processes")
            self._force_cleanup_zombie_processes()
```

        task_executor = TaskExecutor.from_container(container)
        
        # Execute task in isolation
        result = task_executor.execute(task)
        
        return result.to_dict()
        
    except ParallelExecutionError as e:
        # ✅ Erro esperado quando código não foi atualizado para contexto paralelo
        logger = LoggingManager.get_instance()  # Fallback para logging de erro crítico
        logger.error(f"CRITICAL: Code not updated for parallel context: {e}")
        logger.error(f"This error indicates singleton usage without process_config in parallel context")

    except Exception as e:
        # Log error and return failure result
        logger = container.get(LoggingManager).get_logger('process_executor')
        logger.error(f"Task execution failed in process {process_config.process_id}: {e}")
        
        return TaskResult.create_error(task.id, str(e)).to_dict()
    
    finally:
        # ✅ Cleanup environment variable
        if RV_PARALLEL_EXECUTION_ENV_VAR in os.environ:
            del os.environ[RV_PARALLEL_EXECUTION_ENV_VAR]
            
        # Cleanup process resources
        _cleanup_process_resources(container, process_config)

def _register_isolated_services(container: DependencyContainer, 
                               process_config: ProcessConfiguration):
    """Registra serviços isolados no container"""
    
    # Register core services with process configuration
    container.register_singleton(
        LoggingManager, 
        lambda: LoggingManager.get_instance(process_config)
    )
    
    container.register_singleton(
        ErrorHandler,
        lambda: ErrorHandler.get_instance(process_config)  
    )
    
    container.register_singleton(
        EventBus,
        lambda: EventBus.get_instance(process_config)
    )
    
    # Register specialized services
    container.register_singleton(
        PortManager,
        lambda: PortManager()  # Stateless, thread-safe
    )
    
    # Módulos de pre-processamento NÃO precisam de configuração específica
    # pois usam artefatos compartilhados (monitors/, instrumented/, static_analysis/)
    # 
    # Apenas módulos de execução precisam de configuração isolada:
    # - rvandroid-tool server (porta)
    # - Android/Emulator (porta + AVD)
    # - TaskExecutor (results + logs)

def _cleanup_process_resources(container: DependencyContainer, 
                             process_config: ProcessConfiguration):
    """Limpa recursos do processo após execução"""
    try:
        # Release allocated ports
        PortManager.release_process_ports(process_config.process_id)
        
        # Shutdown services gracefully
        event_bus = container.get(EventBus)
        if hasattr(event_bus, 'shutdown'):
            event_bus.shutdown()
            
    except Exception as e:
        # Log cleanup errors but don't fail the process
        print(f"Error during process cleanup: {e}")

class TaskExecutor:
    """Executor de tasks com injeção de dependências"""
    
    def __init__(self, 
                 logging_manager: LoggingManager,
                 error_handler: ErrorHandler, 
                 event_bus: EventBus,
                 **component_configs):
        """
        Initialize com dependências injetadas
        
        Args:
            logging_manager: Instance isolada de logging
            error_handler: Instance isolada de error handling
            event_bus: Instance isolada de event bus
            **component_configs: Configurações dos componentes
        """
        self.logging_manager = logging_manager
        self.error_handler = error_handler
        self.event_bus = event_bus
        self.component_configs = component_configs
        
        # Setup logger
        self.logger = logging_manager.get_logger(
            'rv_platform.execution.executor',
            {CONTEXT_COMPONENT: 'TaskExecutor'}
        )
        
    @classmethod 
    def from_container(cls, container: DependencyContainer) -> 'TaskExecutor':
        """Factory method usando DI container"""
        return cls(
            logging_manager=container.get(LoggingManager),
            error_handler=container.get(ErrorHandler),
            event_bus=container.get(EventBus),
            # Configurações específicas por processo apenas para recursos isolados
            process_config=container.process_config
        )
        
    def execute(self, task: Task) -> TaskResult:
        """Executa task usando dependências isoladas"""
        
        # Use isolated event bus for task events
        self.event_bus.publish_task_event(
            EventType.TASK_STARTED,
            task.id,
            task_config=task.config
        )
        
        try:
            # Execute task components in isolation
            result = self._execute_task_components(task)
            
            # Publish success event
            self.event_bus.publish_task_event(
                EventType.TASK_COMPLETED,
                task.id,
                details={'result': result.summary}
            )
            
            return result
            
        except Exception as e:
            # Handle error with isolated error handler
            self.error_handler.handle_error(e, {
                'component': 'TaskExecutor',
                'operation': 'execute',
                'task_id': task.id
            })
            
            # Publish failure event
            self.event_bus.publish_task_event(
                EventType.TASK_FAILED,
                task.id,
                details={'error': str(e)}
            )
            
            raise

class ExperimentController:
    """Controlador de experimentos com suporte a paralelização"""
    
    def __init__(self, config: ExperimentConfig):
        self.config = config
        
        # Determinar se deve usar execução paralela
        self.use_parallel = self._should_use_parallel_execution()
        
        if self.use_parallel:

        if not self.config.parallel_config.enabled:
            return False
            
        # Check environment variable override
        if os.getenv(ENV_PARALLEL_ENABLED, '').lower() == 'false':
            return False
        
        # Check resource availability
        try:
            port_status = PortManager.get_port_status()
            min_required = min(2, self.config.parallel_config.max_processes)
            
            if (port_status['emulator_ports_available'] < min_required or 
                port_status['server_ports_available'] < min_required):
                self.logger.warning("Insufficient resources for parallel execution, using sequential")
                return False
                
        except Exception as e:
            self.logger.warning(f"Resource check failed, using sequential: {e}")
            return False
        
        return True
    
    def run_experiment(self) -> ExperimentResult:
        """Executa experimento com coordenação de paralelização"""
        
        # Pre-processing phase (sempre sequencial)
        self._run_pre_processing()
        
        # Execution phase (paralela ou sequencial)
        if self.use_parallel:
            execution_results = self._run_parallel_execution()
        else:
            execution_results = self._run_sequential_execution()
        
        # Post-processing phase (sempre sequencial)  
        final_results = self._run_post_processing(execution_results)
        
        return final_results
    
    def _run_pre_processing(self):
        """Pre-processamento sequencial (instrumentação, análise estática, etc.)"""
        # Geração de monitores (uma vez, compartilhado)
        self._generate_monitors()
        
        # Instrumentação de APKs (uma vez, compartilhado)
        self._instrument_apks()
        
        # Análise estática (uma vez, compartilhado)
        self._run_static_analysis()
        
        # Artefatos ficam em diretórios compartilhados:
        # - monitors/ 
        # - instrumented/
        # - static_analysis/
        
    def _run_parallel_execution(self) -> Dict[str, Any]:
        """Execução paralela através do rv-platform"""
        
        # Criar tasks para execução paralela
        tasks = self._create_execution_tasks()
        
        # Delegar para rv-platform com configuração paralela
        # rv-platform usa artefatos compartilhados do pre-processamento
        platform_config = PlatformConfig(
            parallel_config=self.config.parallel_config,
            tools_config=self.config.tools_config,
            base_output_dir=self.config.base_output_dir  # Diretório com artefatos compartilhados
        )
        
        platform = Platform(platform_config)
        return platform.execute_tasks_parallel(tasks)
        
    def _run_sequential_execution(self) -> Dict[str, Any]:
        """Execução sequencial tradicional"""

def get_pending_tasks(self) -> List[Task]:
    """
    Get tasks that are not yet completed, failed, or canceled.

    return [task for task in self.tasks if task.state == TaskState.PENDING]

def resume_experiment_from_directory(self, results_dir: str) -> Tuple[ExperimentConfig, List[Task]]:
    """

    Returns:
        Tuple of (original_experiment_config, pending_tasks)

    # Load experiment metadata and validate

        raise ResumeError(f"No experiment status found in {results_dir}")

    original_config = ExperimentConfig.load_from_status(status_file)
    
    # Load task storage and get pending tasks
    task_storage = TaskStorage(status_file)
    task_storage.load()
    pending_tasks = task_storage.get_pending_tasks()
    
    return original_config, pending_tasks

import multiprocessing  # ✅ NÃO threading

class PortManager:

    _allocated_emulator_ports = _manager.set()
    _allocated_server_ports = _manager.set()

class ProcessReportingEventBus(EventBus):

                 parent_queue: Optional[multiprocessing.Queue] = None):
